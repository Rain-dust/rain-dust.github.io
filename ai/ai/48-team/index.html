    const check = await fetch("/backend-api/accounts/check/v4-2023-04-27", {
        headers: { Authorization: `Bearer ${session.accessToken}` },
    }).then((r) => r.json());
    const accountId = Object.keys(check.accounts || {})[0];
    if (!accountId) throw new Error("找不到 workspace account_id");
    const headers = {
        Authorization: `Bearer ${session.accessToken}`,
        "Content-Type": "application/json",
        "ChatGPT-Account-ID": accountId,
    };
    const res = await fetch(
        `/backend-api/accounts/${accountId}/settings/allow_external_domain_invites`,
        {
            method: "POST",
            headers,
            body: JSON.stringify({ value: true }),
        }
    );
    const data = await res.json().catch(() => ({}));
    console.log("POST status:", res.status, data);
    const identity = await fetch(
        `/backend-api/accounts/${accountId}/identity`,
        { headers }
    ).then((r) => r.json());
    console.log("allow_external_domain_invites =", identity.allow_external_domain_invites);
})();
